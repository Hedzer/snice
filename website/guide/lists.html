Keyed lists

repeat() makes identity explicit, moves existing DOM on reorder, rejects duplicate keys, and renders an empty state without a wrapper.

@element('task-list') class TaskList extends HTMLElement { @property({ type: Array }) items = []; @render() template() { return html` ${repeat(this.items, { key: item => item.id, render: item => html` <task-item label=${item.label} ?done=${item.done} @status-changed=${(e) => this.update(e.detail)} ></task-item> `, empty: () => html`<p>No tasks</p>` })} `; } }
